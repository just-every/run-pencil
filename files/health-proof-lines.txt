    17	async function retained(actionId: string) {
    18	  return Promise.all(['actions', 'action_events', 'action_task_deliveries', 'action_task_attempts', 'action_reminders', 'action_reminder_attempts'].map(table => {
    19	    const condition = table === 'actions' ? 'id=?' : table === 'action_reminder_attempts'
    20	      ? 'reminder_id IN (SELECT id FROM action_reminders WHERE action_id=?)' : 'action_id=?'
    21	    return env.DB.prepare(`SELECT * FROM ${table} WHERE ${condition} ORDER BY 1`).bind(actionId).all().then(result => result.results)
    22	  }))
    23	}

    25	export async function taskFailureHealth(f: Fixture, actionId: string, state: 'failed' | 'unknown') {
    26	  const stored = await env.DB.prepare('SELECT id FROM action_task_attempts WHERE action_id=? ORDER BY ordinal').bind(actionId).all<{id: string}>()
    27	  const before = await retained(actionId), alreadyMocked = vi.isMockFunction(globalThis.fetch), outbound = vi.spyOn(globalThis, 'fetch'), calls = outbound.mock.calls.length
    28	  let report: ActionHealthReport
    29	  try {
    30	    report = await readHealth(f, actionId)
    31	    expect(await readHealth(f, actionId)).toEqual(report)
    32	    expect(outbound).toHaveBeenCalledTimes(calls)
    33	  } finally { if (!alreadyMocked) outbound.mockRestore() }
    34	  expect(await retained(actionId)).toEqual(before)
    35	  expect(report.task.delivery).toMatchObject({ actionId, idempotencyKey: `pencil-action-${actionId}`, state, stale: false })
    36	  expect(report.task.delivery!.attempts.map(item => item.id)).toEqual(stored.results.map(item => item.id))
    37	  expect(report.task.delivery!.attempts[0].explanation).toMatch(state === 'failed' ? /server error/ : /may have created/)
    38	  expect(report.task.repairPlan).toEqual([expect.stringContaining('explicit operator retry')])
    39	  return report
    40	}

    43	export async function brokenActionHealth(receiver: string) {
    44	  await fetch(`${receiver}/__control/reset`, { method: 'POST', body: JSON.stringify({ mode: 'fail-once-500' }) })
    45	  const key = 'synthetic-pencil-health-1'
    46	  const f = await actionsHttp({ TASK_PROVIDER: 'webhook', TASK_WEBHOOK_URL: `${receiver}/tasks`, TASK_API_KEY: key } as Partial<WorkerBindings>)
    47	  const departing = await f.operator('Departing health owner')
    48	  const action = await f.create({ ownerOperatorId: departing.operatorId, rationale: 'synthetic-pencil-private-action' })
    49	  const taskPath = `/api/app/actions/${action.id}/task`, healthPath = `/api/app/actions/${action.id}/health`
    50	  const preview = await (await f.call(taskPath)).json() as {payloadSha256: string}
    51	  expect((await f.call(`${taskPath}/confirm`, { payloadSha256: preview.payloadSha256 })).status).toBe(200)
    52	  await f.services.auth.removeOperator({ workspaceId: f.workspaceId, actorOperatorId: f.lead.operatorId, targetOperatorId: departing.operatorId })
    53	  const foreign = await actionsHttp()
    54	  const revoked = await f.operator('Revoked health reader')
    55	  await env.DB.prepare('DELETE FROM oa_sessions WHERE id=?').bind(revoked.sessionId).run()
    56	  const before = await retained(action.id), alreadyMocked = vi.isMockFunction(globalThis.fetch), outbound = vi.spyOn(globalThis, 'fetch'), calls = outbound.mock.calls.length
    57	  let report: ActionHealthReport
    58	  try {
    59	    report = await readHealth(f, action.id)
    60	    expect(await readHealth(f, action.id)).toEqual(report)
    61	    for (const [call, expected] of [
    62	      [() => f.call(healthPath, undefined, { auth: false }), 401],
    63	      [() => foreign.call(healthPath), 404],
    64	      [() => f.as(revoked.token)(healthPath), 401],
    65	    ] as const) {
    66	      const denied = await call()
    67	      expect(denied.status).toBe(expected)
    68	      const body = await denied.text()
    69	      for (const privateValue of [action.id, f.workspaceId, 'synthetic-pencil-private-action', `pencil-action-${action.id}`]) expect(body).not.toContain(privateValue)
    70	    }
    71	    expect(outbound).toHaveBeenCalledTimes(calls)
    72	  } finally { if (!alreadyMocked) outbound.mockRestore() }
    73	  expect(await retained(action.id)).toEqual(before)
    74	  expect(report.task.delivery).toMatchObject({ actionId: action.id, state: 'failed', stale: true })
    75	  const attempt = await env.DB.prepare('SELECT id FROM action_task_attempts WHERE action_id=?').bind(action.id).first<{id: string}>()
    76	  expect(report.task.delivery!.attempts).toEqual([expect.objectContaining({ id: attempt!.id, state: 'failed', httpStatus: 500 })])
    77	  expect(report.reminders).toMatchObject({ actionId: action.id, state: 'no_owner', explanation: expect.stringContaining('no current owner'), repairPlan: [expect.stringContaining('operator must choose')] })
    78	  for (const value of [key, 'synthetic-pencil-private-action', receiver]) expect(JSON.stringify(report)).not.toContain(value)
    79	  // Repairing the owner changes the payload: a retry cannot approve that change.
    80	  expect((await f.call(`/api/app/actions/${action.id}`, { ownerOperatorId: f.owner.operatorId, expectedVersion: action.version }, { method: 'PATCH' })).status).toBe(200)
    81	  expect((await f.call(`${taskPath}/retry`, {})).status).toBe(409)
    82	  expect((await readHealth(f, action.id)).task.repairPlan).toEqual([expect.stringContaining('explicitly confirm it again')])
    83	  const state = await (await fetch(`${receiver}/__control/state`)).json() as {requests: unknown[]}
    84	  expect(state.requests).toHaveLength(1)
    85	  const fresh = await (await f.call(taskPath)).json() as {payloadSha256: string}
    86	  expect(fresh.payloadSha256).not.toBe(preview.payloadSha256)
    87	  expect((await (await f.call(`${taskPath}/confirm`, {payloadSha256: fresh.payloadSha256})).json()).delivery.state).toBe('confirmed')
    88	  const repaired = await readHealth(f, action.id)
    89	  expect(repaired.task.delivery).toMatchObject({state: 'confirmed', attempts: [expect.objectContaining({id: attempt!.id})]})
    90	  expect(repaired.task.repairPlan).toEqual([])
    91	  expect(repaired.reminders.state).toBe('suppressed')
    92	}

    94	export async function reminderFailureHealth(f: Fixture, unknownActionId: string, failedActionId: string) {
    95	  // Retained sent history can push an old failure off the normal detail page; health must still include it.
    96	  await env.DB.batch(Array.from({length: 31}, (_, index) => env.DB.prepare(`INSERT INTO action_reminders
    97	    (id,action_id,operator_id,kind,due_date,review_date,status,idempotency_key,attempts,available_at,created_at,updated_at)
    98	    SELECT ?,action_id,operator_id,'overdue',?,review_date,'sent',?,1,available_at,created_at,updated_at
    99	    FROM action_reminders WHERE action_id=? AND status='failed' LIMIT 1`)
   100	    .bind(crypto.randomUUID(), new Date(Date.parse('2027-01-01T00:00:00Z') + index * 86400000).toISOString().slice(0,10), crypto.randomUUID(), failedActionId)))
   101	  const recent = await (await f.call(`/api/app/actions/${failedActionId}`)).json() as {reminders: {deliveries: Array<{status: string}>}}
   102	  expect(recent.reminders.deliveries.length).toBeGreaterThan(0)
   103	  expect(recent.reminders.deliveries.every(item => item.status === 'sent')).toBe(true)
   104	  for (const [actionId, state] of [[unknownActionId, 'unknown'], [failedActionId, 'failed']] as const) {
   105	    const occurrences = await env.DB.prepare('SELECT id FROM action_reminders WHERE action_id=? AND status=?').bind(actionId, state).all<{id: string}>()
   106	    expect(occurrences.results).toHaveLength(1)
   107	    const before = await retained(actionId), alreadyMocked = vi.isMockFunction(globalThis.fetch), outbound = vi.spyOn(globalThis, 'fetch'), calls = outbound.mock.calls.length
   108	    try {
   109	      const report = await readHealth(f, actionId)
   110	      expect(report.reminders.occurrences).toEqual([expect.objectContaining({id: occurrences.results[0].id, state, attempts: 1, explanation: expect.stringContaining(state === 'unknown' ? 'may have succeeded' : 'failed'), repairPlan: [expect.stringContaining(state === 'unknown' ? 'unresent' : 'does not replay')]})])
   111	      expect(await readHealth(f, actionId)).toEqual(report)
   112	      expect(outbound).toHaveBeenCalledTimes(calls)
   113	    } finally { if (!alreadyMocked) outbound.mockRestore() }
   114	    expect(await retained(actionId)).toEqual(before)
   115	  }
   116	  // Mail configuration explains schedule failure without exposing the configured sender or credentials.
   117	  const sender = ['synthetic-pencil-sender', 'example.test'].join(String.fromCharCode(64))
   118	  const origin = 'https://forms.example.test'

   122	  } as WorkerBindings))
   123	  const response = await unavailable.request(`${origin}/api/app/actions/${failedActionId}/health`, {headers: {Cookie: `__Host-operator_session=${f.lead.token}`}})
   124	  expect(response.status).toBe(200)
   125	  const report = await response.json() as ActionHealthReport
   126	  expect(report.reminders).toMatchObject({actionId: failedActionId, state: 'unavailable', repairPlan: [expect.stringContaining('repair mail configuration')]})
   127	  for (const value of ['synthetic-pencil-health-2', sender]) expect(JSON.stringify(report)).not.toContain(value)
   128	}
