  12 export async function readHealth(f: Fixture, actionId: string) {
  13   const response = await f.call(`/api/app/actions/${actionId}/health`)
  14   expect(response.status).toBe(200)
  15   expect(response.headers.get('cache-control')).toBe('no-store')
  16   return await response.json() as ActionHealthReport
  17 }

  19 async function retained(actionId: string) {
  20   return Promise.all(['actions', 'action_events', 'action_task_deliveries', 'action_task_attempts', 'action_reminders', 'action_reminder_attempts'].map(table => {
  21     const condition = table === 'actions' ? 'id=?' : table === 'action_reminder_attempts'
  22       ? 'reminder_id IN (SELECT id FROM action_reminders WHERE action_id=?)' : 'action_id=?'
  23     return env.DB.prepare(`SELECT * FROM ${table} WHERE ${condition} ORDER BY 1`).bind(actionId).all().then(result => result.results)
  24   }))
  25 }

  27 export async function taskFailureHealth(f: Fixture, actionId: string, state: 'failed' | 'unknown') {
  28   const stored = await env.DB.prepare('SELECT id FROM action_task_attempts WHERE action_id=? ORDER BY ordinal').bind(actionId).all<{id: string}>()
  29   const before = await retained(actionId), alreadyMocked = vi.isMockFunction(globalThis.fetch), outbound = vi.spyOn(globalThis, 'fetch'), calls = outbound.mock.calls.length
  30   let report: ActionHealthReport
  31   try {
  32     report = await readHealth(f, actionId)
  33     expect(await readHealth(f, actionId)).toEqual(report)
  34     expect(outbound).toHaveBeenCalledTimes(calls)
  35   } finally { if (!alreadyMocked) outbound.mockRestore() }
  36   expect(await retained(actionId)).toEqual(before)
  37   expect(report.task.delivery).toMatchObject({ actionId, idempotencyKey: `pencil-action-${actionId}`, state, stale: false })
  38   expect(report.task.delivery!.attempts.map(item => item.id)).toEqual(stored.results.map(item => item.id))
  39   expect(report.task.delivery!.attempts[0].explanation).toMatch(state === 'failed' ? /server error/ : /may have created/)
  40   expect(report.task.repairPlan).toEqual([expect.stringContaining('explicit operator retry')])
  41   return report
  42 }

  53     .bind(nativeOwnerId,nativeOwnerId,departing.operatorId).run()

  56   const taskPath = `/api/app/actions/${action.id}/task`, healthPath = `/api/app/actions/${action.id}/health`
  57   const preview = await (await f.call(taskPath)).json() as {payloadSha256: string}
  58   expect((await f.call(`${taskPath}/confirm`, { payloadSha256: preview.payloadSha256 })).status).toBe(200)
  59   expect((await f.call(`/api/app/operators/${nativeOwnerId}`, {}, {method: 'DELETE'})).status).toBe(204)
  60   const foreign = await actionsHttp()
  61   const revoked = await f.operator('Revoked health reader')
  62   await env.DB.prepare('DELETE FROM oa_sessions WHERE id=?').bind(revoked.sessionId).run()
  63   const before = await retained(action.id), alreadyMocked = vi.isMockFunction(globalThis.fetch), outbound = vi.spyOn(globalThis, 'fetch'), calls = outbound.mock.calls.length
  64   let report: ActionHealthReport
  65   try {
  66     report = await readHealth(f, action.id)
  67     expect(await readHealth(f, action.id)).toEqual(report)
  68     for (const [call, expected] of [
  69       [() => f.call(healthPath, undefined, { auth: false }), 401],
  70       [() => foreign.call(healthPath), 404],
  71       [() => f.as(revoked.token)(healthPath), 401],
  72       [() => f.as(departing.token)(healthPath), 401],
  73     ] as const) {
  74       const denied = await call()

  76       const body = await denied.text()
  77       for (const privateValue of [action.id, f.workspaceId, 'synthetic-pencil-private-action', `pencil-action-${action.id}`]) expect(body).not.toContain(privateValue)
  78     }
  79     expect(outbound).toHaveBeenCalledTimes(calls)

  81   expect(await retained(action.id)).toEqual(before)
  82   expect(report.task.delivery).toMatchObject({ actionId: action.id, state: 'failed', stale: true })

  85   expect(report.reminders).toMatchObject({ actionId: action.id, state: 'no_owner', explanation: expect.stringContaining('no current owner'), repairPlan: [expect.stringContaining('operator must choose')] })
  86   for (const value of [key, 'synthetic-pencil-private-action', receiver]) expect(JSON.stringify(report)).not.toContain(value)
  87   // Repairing the owner changes the payload: a retry cannot approve that change.
  88   expect((await f.call(`/api/app/actions/${action.id}`, { ownerOperatorId: f.owner.operatorId, expectedVersion: action.version }, { method: 'PATCH' })).status).toBe(200)
  89   expect((await f.call(`${taskPath}/retry`, {})).status).toBe(409)
  90   expect((await readHealth(f, action.id)).task.repairPlan).toEqual([expect.stringContaining('explicitly confirm it again')])
  91   const state = await (await fetch(`${receiver}/__control/state`)).json() as {requests: unknown[]}
  92   expect(state.requests).toHaveLength(1)
  93   const fresh = await (await f.call(taskPath)).json() as {payloadSha256: string}

  96   const beforeConfirmation = await (await fetch(`${receiver}/__control/state`)).json() as {requests: unknown[]}
  97   expect(beforeConfirmation.requests).toHaveLength(1)
  98   expect((await (await f.call(`${taskPath}/confirm`, {payloadSha256: fresh.payloadSha256})).json()).delivery.state).toBe('confirmed')
  99   const repaired = await readHealth(f, action.id)
